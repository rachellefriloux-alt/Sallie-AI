# Sallie Brain — runtime dependencies (Phase 1 minimum).
# Subsequent phases will add: qdrant-client, ollama, sentence-transformers,
# numpy, pandas, etc. (see legacy/Sallie/requirements.txt for the full list).

# fastapi >= 0.122 allows starlette 0.49.x, which is required to pick up
# the patched starlette (>= 0.49.1) that fixes the Range-header DoS.
fastapi==0.130.0
starlette==0.49.1
uvicorn[standard]==0.30.6
pydantic==2.9.2
# python-multipart is intentionally omitted: the brain does not accept
# multipart uploads. Add it back (>= 0.0.22) only if upload routes appear.

# httpx is used both for outbound calls (e.g. the knowledge-service client
# in app.clients.knowledge) and by FastAPI's TestClient at test time.
httpx==0.27.2

# OpenAI SDK — used by app.synthesis.composer.GitHubModelsResponder to
# call the GitHub Models OpenAI-compatible endpoint
# (https://models.github.ai/inference). Only loaded when the responder is
# selected via SALLIE_RESPONDER=github_models, so the brain still boots
# without a GITHUB_TOKEN.
openai==1.54.0

# Auth — Phase 1 item 22.
# passlib provides a stable wrapper around bcrypt; we pin bcrypt 4.x to
# avoid the silent ABI break that landed in bcrypt 4.0 vs. 3.x. PyJWT 2.x
# is the canonical Python JWT library.
passlib[bcrypt]==1.7.4
bcrypt==4.2.0
pyjwt==2.9.0
# email-validator powers Pydantic's EmailStr — used by the auth route DTOs
# to reject obviously-malformed addresses before they hit the DB.
email-validator==2.2.0

# Persistence — Phase 1.1.
# SQLAlchemy 2.x async + the two drivers we ship: asyncpg for production
# (Postgres) and aiosqlite for tests (in-memory). The ORM models in
# app/db/models.py use portable types so the same models work on both.
sqlalchemy[asyncio]==2.0.36
asyncpg==0.30.0
aiosqlite==0.20.0

# Dev / test
pytest==8.3.3
pytest-asyncio==0.24.0
